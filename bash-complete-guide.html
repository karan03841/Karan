scan_port() {
    local host=$1    # local = function ke andar hi
    local port=$2

    # Timeout ke saath port check
    if timeout 1 bash -c "echo >/dev/tcp/$host/$port" 2>/dev/null; then
        success "Port $port is OPEN on $host"
        return 0
    else
        error "Port $port is CLOSED"
        return 1
    fi
}

# Use karo

#!/bin/bash
# ================================================================
# AUTO RECON SCRIPT — Karan's OSCP Tool
# Usage: ./recon.sh 10.10.10.10
# ================================================================

# Colors
RED='\e[31m'; GREEN='\e[32m'; YELLOW='\e[33m'
CYAN='\e[36m'; BOLD='\e[1m'; NC='\e[0m'

banner() {
    echo -e "${GREEN}"
    echo "╔══════════════════════════════════════╗"
    echo "║     AUTO RECON — KARAN v1.0          ║"
    echo "║     HTB/OSCP Automation              ║"
    echo "╚══════════════════════════════════════╝"
    echo -e "${NC}"
}

info()    { echo -e "${CYAN}[*]${NC} $1"; }
success() { echo -e "${GREEN}[+]${NC} $1"; }
warn()    { echo -e "${YELLOW}[!]${NC} $1"; }
error()   { echo -e "${RED}[-]${NC} $1"; }
section() { echo -e "\n${BOLD}${YELLOW}━━━ $1 ━━━${NC}"; }

if [ $# -lt 1 ]; then
    error "Usage: $0 <IP> [output_dir]"
    exit 1
fi

TARGET=$1
OUTDIR=${2:-"recon_$TARGET"}
mkdir -p "$OUTDIR"

banner
info "Target: $TARGET"
info "Output: $OUTDIR"
info "Start Time: $(date)"

# 1. Ping check
section "PING CHECK"
if ping -c2 -W2 $TARGET &>/dev/null; then
    success "$TARGET is UP"
else
    warn "$TARGET not responding to ping (might be filtered)"
fi

# 2. Quick port scan
section "QUICK PORT SCAN"
info "Running fast scan..."
nmap -T4 --open -p- $TARGET -oN "$OUTDIR/ports.txt" 2>/dev/null
OPEN_PORTS=$(grep "open" "$OUTDIR/ports.txt" | awk '{print $1}' | cut -d/ -f1 | tr '\n' ',')
success "Open ports: $OPEN_PORTS"

# 3. Service scan
section "SERVICE DETECTION"
if [ -n "$OPEN_PORTS" ]; then
    info "Running service scan on: $OPEN_PORTS"
    nmap -sV -sC -p${OPEN_PORTS%,} $TARGET -oN "$OUTDIR/services.txt" 2>/dev/null
    success "Service scan complete"
fi

# 4. HTTP check
section "WEB CHECK"
for port in 80 443 8080 8443 8000 8888; do
    if echo "$OPEN_PORTS" | grep -q "$port"; then
        info "Web service on port $port found!"
        curl -sk -I "http://$TARGET:$port" | head -5 | tee -a "$OUTDIR/web_$port.txt"
        # Gobuster
        info "Running gobuster on port $port..."
        gobuster dir -u "http://$TARGET:$port" \
            -w /usr/share/wordlists/dirbuster/directory-list-2.3-medium.txt \
            -o "$OUTDIR/gobuster_$port.txt" -q 2>/dev/null &
    fi
done

# 5. SMB check
if echo "$OPEN_PORTS" | grep -q "445"; then
    section "SMB ENUMERATION"
    info "SMB detected — running enum4linux..."
    enum4linux -a $TARGET > "$OUTDIR/smb.txt" 2>/dev/null &
fi

# Wait for background jobs
wait
section "SCAN COMPLETE"
success "All results in: $OUTDIR/"
ls -la "$OUTDIR/"
