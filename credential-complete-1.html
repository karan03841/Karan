    Route            Technique

RUN KARNE KA TARIKA (apne Ubuntu pe):
    pip3 install flask --break-system-packages

# attack karna bahut slow/useless hoga.

<h2>Spray-Login Lab (Technique #3)</h2>

Ek hi password ko sab usernames ke against try karo (password spraying).</p>

# ---------------------------------------------------------------------
# CAPTCHA dikhta hai, lekin server kabhi bhi uski value check nahi karta —
# sirf ek hidden field match karta hai jo khud response me bhej deta hai.

CAPTCHA_FORM = """
<h2>CAPTCHA-Login Lab (Technique #12)</h2>
<p>Yeh form ek CAPTCHA dikhata hai (3 + 4 = ?) lekin dekho ke server
usko kaise (nahi) validate karta hai — View Source / Burp response check karo.</p>

"""

CAPTCHA_ATTEMPTS = []

@app.route("/captcha-login", methods=["GET", "POST"])
def captcha_login():
    if request.method == "GET":
        return CAPTCHA_FORM.replace("{len(CAPTCHA_ATTEMPTS)}", str(len(CAPTCHA_ATTEMPTS)))
    u = request.form.get("username", "")
    p = request.form.get("password", "")
    CAPTCHA_ATTEMPTS.append((u, p))
    # BUG: server never actually checks the captcha field against a
    # server-side-stored answer — it silently ignores it entirely.
    if u == "admin" and p == "RedTeam2024!":
        return "Login successful! (captcha was never actually enforced)", 200
    return "Invalid username or password", 401


# ---------------------------------------------------------------------
# ROUTE 5  —  Technique #18: Low-and-Slow Brute Force
# ---------------------------------------------------------------------
# Rate-limiter: same IP se 10-second window me 3 se zyada attempts
# aaye to 15 seconds ke liye 429 block ho jata hai. Fast Hydra
# (-t 16, no -w) is route pe turant block ho jayega. Slow Hydra
# (-t 1 -w 4) kabhi threshold cross nahi karega aur successfully
# login find kar lega.

RATE_WINDOW = 10      # seconds
RATE_MAX = 3           # max attempts per window
BLOCK_SECONDS = 15

_attempts = defaultdict(list)   # ip -> [timestamps]
_blocked_until = {}              # ip -> unix timestamp

SLOW_PASSWORD = "Fortress99"

SLOW_FORM = """
<h2>Low-and-Slow Lab (Technique #18)</h2>
<p>Is route pe ek real rate-limiter hai: same IP se 10 second window me
3 se zyada attempts = 15 second ka block (HTTP 429).</p>
<p>Fast brute-force try karo pehle (dekho block hota hai), phir slow
(delay ke saath) try karo (dekho block nahi hota).</p>
<form method="POST">
  Password (user is fixed: admin): <input name="password"><br>
  <button type="submit">Login</button>
</form>
<p>Hint fast (blocks): <code>hydra -l admin -P rockyou.txt -t 16 127.0.0.1 -s 5000 http-post-form "/slow-login:password=^PASS^:F=Invalid"</code></p>
<p>Hint slow (bypasses): <code>hydra -l admin -P rockyou.txt -t 1 -w 4 127.0.0.1 -s 5000 http-post-form "/slow-login:password=^PASS^:F=Invalid"</code></p>
"""

@app.route("/slow-login", methods=["GET", "POST"])
def slow_login():
    if request.method == "GET":
        return SLOW_FORM
    ip = request.remote_addr
    now = time.time()

    if ip in _blocked_until and now < _blocked_until[ip]:
        remaining = round(_blocked_until[ip] - now, 1)
        return jsonify({"error": f"Too many attempts. Blocked for {remaining}s more."}), 429

    _attempts[ip] = [t for t in _attempts[ip] if now - t < RATE_WINDOW]
    _attempts[ip].append(now)

    if len(_attempts[ip]) > RATE_MAX:
        _blocked_until[ip] = now + BLOCK_SECONDS
        return jsonify({"error": f"Rate limit hit. Blocked for {BLOCK_SECONDS}s."}), 429

    p = request.form.get("password", "")
    if p == SLOW_PASSWORD:
        return "Login successful! (you evaded the rate-limit by going slow)", 200
    return "Invalid username or password", 401

</ul>
"""
