Tool 1 — Nmap (Port Scanner)

# Already installed hai! Try karo:
nmap google.com            ← Basic scan
nmap -sV google.com        ← Service version
nmap -p 80,443 google.com  ← Specific ports
nmap --top-ports 100 target.com

# Results file mein save karo
nmap target.com -oN scan.txt
cat scan.txt

Tool 2 — curl (Sabka Baap)

# Ye pehle se hai — practice karo
curl https://target.com           ← GET
curl -I https://target.com        ← Sirf headers
curl -v https://target.com        ← Verbose
curl -X POST -d "a=b" target.com  ← POST
curl -H "Authorization: Bearer X" target.com
curl -o file.html target.com      ← Save karo

# Loop se IDOR test!
for i in $(seq 1 10); do
  echo "ID $i:"
  curl -s "https://httpbin.org/get?id=$i"
done

Tool 3 — Go Tools Install (Subfinder, httpx, nuclei)
